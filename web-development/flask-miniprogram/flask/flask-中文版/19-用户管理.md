# 19 用户管理

本章实现二手房项目的用户模块：注册、登录状态的维持（cookie）、退出登录、个人中心页面（收藏与浏览记录）、收藏 / 取消收藏、清空浏览记录，以及编辑用户信息。所有视图都注册在 `app_user` 蓝图上，接口统一返回 JSON（`{'valid': ..., 'msg': ...}` 或 `{'ok': ...}`）。

**关于密码存储**：密码在数据库中保存的是**密文**（加密后的文本），加密是**不可逆**的（不能解密还原出原文）；相同内容加密出来的结果相同，因此校验密码时直接比较密文是否一致即可。

## 19.1 退出登录

登录状态通过 cookie 中的 `name`（用户名）判断，退出登录就是**清除 cookie 中的 `name`**：

```python
# 退出登录  {'valid': '1', 'msg': '退出登录成功'}  {'valid': '0', 'msg': '未登录'}
@app_user.route('/logout', methods=['GET'])
def logout():
    # 获取cookie中的用户信息
    name = request.cookies.get('name')

    if name:
        # 构建json响应对象
        res = jsonify({'valid': '1', 'msg': '退出登录成功'})

        # 删除cookie --》 name
        res.delete_cookie('name')
        return res
    else:
        return jsonify({'valid': '0', 'msg': '未登录'})
```

注意 `delete_cookie` 必须作用在**响应对象**上，所以要先构建 `jsonify` 响应、删除 cookie 后再返回。

## 19.2 登录

前端以 POST 表单提交 `username` 和 `password`，后端查询用户、校验密码，成功后把用户名写入 cookie（有效期 1 天）：

```python
# 登录  username  password  ----> post ---> form
@app_user.route('/login', methods=['POST'])
def login():
    username = request.form.get('username')
    password = request.form.get('password')

    # 通过用户名 获取 用户对象
    user = User.query.filter(User.name == username).first()

    # 判断用户数据是否存在
    if user:
        # 密码校验
        if user.password == password:
            # 响应对象构建 ---》 成功
            res = jsonify({'valid': '1', 'msg': user.name})
            # 设置cookie
            res.set_cookie('name', user.name, max_age=60 * 60 * 24)
            return res
        else:
            return jsonify({'valid': '0', 'msg': '密码错误'})
    else:
        return jsonify({'valid': '0', 'msg': '用户名错误'})
```

之后所有需要登录态的接口，都通过 `request.cookies.get('name')` 获取当前登录用户。

## 19.3 用户界面（个人中心）

个人中心页面展示用户信息、**收藏的房源**和**浏览过的房源**。收藏与浏览记录都存在 `User` 表的字符串字段里（`collect_id`、`seen_id`），格式为逗号分隔的房源 id，如 `"234,345"`。取数据时先 `split` 成列表，再逐个查询房源对象：

```python
# 用户界面
@app_user.route('/user/<name>', methods=['GET'])
def user(name):
    user = User.query.filter(User.name == name).first()

    if user:
        # 收藏数据  User ---> collect_id  收藏房源的id ---》 string类型
        # print(user.collect_id)  # 234,345
        collect_id_str = user.collect_id
        collect_house_list = []  # 收藏房源列表

        if collect_id_str:
            # 如何获取房源数据？
            # ----- 1. 分割 split
            collect_id_list = collect_id_str.split(',')  # ['234', '345']

            # ----- 2. 根据 房源id 查询房源数据
            for collect_id in collect_id_list:
                house = House.query.get(collect_id)
                collect_house_list.append(house)

        seen_house_list = []  # 浏览房源列表
        seen_id_str = user.seen_id
        if seen_id_str:
            seen_id_list = seen_id_str.split(',')
            for seen_id in seen_id_list:
                house = House.query.get(seen_id)
                seen_house_list.append(house)

        return render_template('user_page.html',
                               user=user,
                               collect_house_list=collect_house_list,
                               seen_house_list=seen_house_list)
    else:
        return redirect(url_for('app_index.index'))
```

用户不存在时直接重定向回首页。

## 19.4 取消收藏

收藏数据存在 `User.collect_id`（如 `"234,345"`）。取消收藏的流程：找到对应房源 id 并从字符串中删除，再**按照原格式**（逗号分隔字符串）存回数据库：

```python
# 收藏数据  ----》 User ---》 collect_id ---》 "234,345"
# 取消收藏：用户对象 ---》 collect_id ---》 找到对应的房源id 删除 ---》
# 再次按照原格式存储更新后的数据
@app_user.route('/collect_off', methods=['POST'])
def collect_off():  # 需要传入取消房源的id
    name = request.form.get('user_name')
    hid = request.form.get('house_id')

    # 用户名 ---》 用户对象 ---》 当前登录用户的（cookie ---》 name）
    if not name or not request.cookies.get('name') or name != request.cookies.get('name'):
        return jsonify({'valid': '0', 'msg': '只能操作自己的数据'})

    # 获取用户对象数据
    user = User.query.filter(User.name == name).first()

    # 收藏房源数据id --- str
    collect_id_str = user.collect_id

    # 收藏房源数据id --- list
    collect_id_list = collect_id_str.split(',')

    # 判断 取消收藏房源id 是否存在于 collect_id_list
    if hid in collect_id_list:
        # 删除指定房源id
        collect_id_list.remove(hid)

        # 重新拼接（格式化）---》 collect_id ---》 str
        new_collect_id_str = ','.join(collect_id_list)

        user.collect_id = new_collect_id_str

        # db.session.add(user)
        db.session.commit()

        res = jsonify({'valid': '1', 'msg': "删除成功"})
        return res
    else:
        res = jsonify({'valid': '0', 'msg': "删除失败"})
        return res
```

要点：

- **权限校验**：表单提交的 `user_name` 必须与 cookie 中的 `name` 一致，只能操作自己的数据。
- 修改的是已查询出的模型对象属性，直接 `db.session.commit()` 即可，无需 `add`。

## 19.5 收藏房源

前端点击收藏后访问带房源 id 的 URL，后端从 cookie 拿到当前用户，判断房源 id 是否已在收藏数据中：没有收藏则添加，已收藏则直接返回提示：

```python
# 收藏
@app_user.route('/add/collection/<int:hid>', methods=['GET'])
def add_collect_id(hid):  # hid --- int
    name = request.cookies.get('name')

    if name:
        user = User.query.filter(User.name == name).first()
        collect_id_str = user.collect_id

        if collect_id_str:
            collect_id_list = collect_id_str.split(',')  # ['234', '345']

            if str(hid) in collect_id_list:
                return jsonify({'valid': '1', 'msg': "已经收藏过了"})
            else:
                # 添加房源id（记得转字符串类型）到 collect_id_list
                collect_id_list.append(str(hid))

                # 重新格式化
                new_collect_id_str = ','.join(collect_id_list)

                user.collect_id = new_collect_id_str
                db.session.commit()
                return jsonify({'valid': '1', 'msg': "收藏成功"})
        else:  # 不存在收藏数据
            # 直接新建收藏数据存入
            user.collect_id = str(hid)
            db.session.commit()
            return jsonify({'valid': '1', 'msg': "收藏成功"})
    else:
        res = jsonify({'valid': '0', 'msg': "请登录后再收藏"})
        return res
```

注意 URL 转换器得到的是 `int` 类型的 `hid`，与字符串列表比较、拼接前要先 `str(hid)`；首次收藏时 `collect_id` 为空，直接赋值即可。

## 19.6 删除浏览记录

清空浏览记录即把 `user.seen_id` 置为空字符串，同样要做登录用户匹配校验：

```python
# 清空浏览记录 ---》 user_name
@app_user.route('/del_record', methods=['POST'])
def del_record():
    name = request.form.get('user_name')

    if not name or not request.cookies.get('name') or name != request.cookies.get('name'):
        return jsonify({'valid': '0', 'msg': "禁止非法操作"})

    user = User.query.filter(User.name == name).first()

    seen_id_str = user.seen_id

    if seen_id_str:
        user.seen_id = ""
        db.session.commit()

        return jsonify({'valid': '1', 'msg': "浏览记录已清空"})
    else:
        return jsonify({'valid': '0', 'msg': "暂无信息可清空"})
```

## 19.7 编辑用户信息

用一个视图统一处理用户名、住址、密码、邮箱四种修改，通过 URL 中的 `<option>` 区分，统一以 `{'ok': 1}` / `{'ok': 0}` 表示成功或失败：

| 请求 URL | 作用 | 表单参数 |
| --- | --- | --- |
| `/modify/userinfo/name` | 编辑用户名 | `name` 新名称、`y_name` 原名称 |
| `/modify/userinfo/addr` | 编辑住址 | `addr` 新地址、`y_name` 用户名 |
| `/modify/userinfo/pd` | 编辑密码 | `pd` 新密码、`y_name` 用户名 |
| `/modify/userinfo/email` | 编辑邮箱 | `email` 新邮箱、`y_name` 用户名 |

每个分支的校验逻辑一致：参数是否齐全（`all([...])`）→ cookie 未过期且 `y_name` 与 cookie 中的 `name` 一致（只能改自己的数据）→ 查询用户并更新。**修改用户名后还要同步更新 cookie 中的 `name`**：

```python
# 编辑用户信息 ---》 POST
# /modify/userinfo/<option>  ----> {'ok': 1}  {'ok': 0}
@app_user.route('/modify/userinfo/<option>', methods=['POST'])
def modify_info(option):
    if option == "name":
        # name 新名称  y_name 原名称
        y_name = request.form.get('y_name')
        name = request.form.get('name')

        # 数据是否获取到
        if not all([y_name, name]):
            return jsonify({'ok': 0})

        # 用户的cookie中name没过期，并且 原名称 与 cookie中name 一致（只能自己改自己的账号数据）
        if not request.cookies.get('name') or y_name != request.cookies.get('name'):
            return jsonify({'ok': 0})

        # 原用户名获取用户数据
        user = User.query.filter(User.name == y_name).first()

        if user:
            # 更新数据
            user.name = name
            db.session.commit()

            # 用户名变更 ---》 cookie 中的 name 也要更新
            res = jsonify({'ok': 1})
            res.set_cookie('name', user.name, max_age=60 * 60 * 24)
            return res
        else:
            return jsonify({'ok': 0})

    elif option == "addr":
        y_name = request.form.get('y_name')
        addr = request.form.get('addr')

        if not all([y_name, addr]):
            return jsonify({'ok': 0})

        if not request.cookies.get('name') or y_name != request.cookies.get('name'):
            return jsonify({'ok': 0})

        user = User.query.filter(User.name == y_name).first()

        if user:
            user.addr = addr
            db.session.commit()
            return jsonify({'ok': 1})
        else:
            return jsonify({'ok': 0})

    elif option == "pd":
        y_name = request.form.get('y_name')
        pd = request.form.get('pd')

        if not all([y_name, pd]):
            return jsonify({'ok': 0})

        if not request.cookies.get('name') or y_name != request.cookies.get('name'):
            return jsonify({'ok': 0})

        user = User.query.filter(User.name == y_name).first()

        if user:
            user.password = pd
            db.session.commit()
            return jsonify({'ok': 1})
        else:
            return jsonify({'ok': 0})

    elif option == "email":
        y_name = request.form.get('y_name')
        email = request.form.get('email')

        if not all([y_name, email]):
            return jsonify({'ok': 0})

        if not request.cookies.get('name') or y_name != request.cookies.get('name'):
            return jsonify({'ok': 0})

        user = User.query.filter(User.name == y_name).first()

        if user:
            user.email = email
            db.session.commit()
            return jsonify({'ok': 1})
        else:
            return jsonify({'ok': 0})
```

注册接口接收 `username`、`password`、`email` 三个表单字段，先检查用户名是否已存在，不存在则写入数据库，并同时完成登录标记（cookie 写入 `name`，效果与登录成功一致）：

```python
# 注册： username  password  email  ---》post请求  ---》 request.form
#       响应： {'valid': '1', 'msg': '用户名'}    {'valid': '0', 'msg': '错误信息'}
@app_user.route('/register', methods=['POST'])
def register():
    # 获取用户注册数据
    username = request.form.get('username')
    password = request.form.get('password')
    email = request.form.get('email')

    # 查询数据库  ---》 用户名是否存在
    result = User.query.filter(User.name == username).all()

    if len(result) == 0:
        user = User(name=username, password=password, email=email)
        db.session.add(user)
        db.session.commit()

        # 利用cookie完成我们的登录标记  ---》 用户名
        res = jsonify({'valid': '1', 'msg': user.name})
        res.set_cookie('name', user.name, max_age=60 * 60 * 24)
        return res
    else:
        return jsonify({'valid': '0', 'msg': '用户名已存在'})
```

注册成功返回 `{'valid': '1', 'msg': 用户名}` 并种下 cookie，前端据此直接进入已登录状态；用户名已存在时返回 `{'valid': '0', 'msg': '用户名已存在'}`。
